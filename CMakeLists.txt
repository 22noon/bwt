cmake_minimum_required(VERSION 3.16)
project(fm_range_bwa_round1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_BENCHMARKS "Build standalone benchmark executables" OFF)
option(BUILD_SINGLETON_INLINE_BENCHMARKS "Build isolated exact-search singleton inlining benchmarks" OFF)
set(BWA_API_DIR "${CMAKE_CURRENT_SOURCE_DIR}/API" CACHE PATH "BWA API source directory")
set(HTSLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/htslib-1.24" CACHE PATH "HTSlib source/install prefix")
if(NOT EXISTS "${BWA_API_DIR}/bwt.h")
  message(FATAL_ERROR "BWA API directory missing bwt.h: ${BWA_API_DIR}")
endif()
find_package(ZLIB REQUIRED)
find_path(HTSLIB_INCLUDE_DIR NAMES htslib/sam.h PATHS "${HTSLIB_ROOT}" "${HTSLIB_ROOT}/include" NO_DEFAULT_PATH)
find_library(HTSLIB_LIBRARY NAMES hts PATHS "${HTSLIB_ROOT}" "${HTSLIB_ROOT}/.libs" "${HTSLIB_ROOT}/lib" "${HTSLIB_ROOT}/lib64" NO_DEFAULT_PATH)
if(NOT HTSLIB_INCLUDE_DIR OR NOT HTSLIB_LIBRARY)
  message(FATAL_ERROR "HTSlib 1.24 headers/library not found under ${HTSLIB_ROOT}. Build/install HTSlib there, or set -DHTSLIB_ROOT=<prefix>.")
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  set(ENV{PKG_CONFIG_PATH} "${HTSLIB_ROOT}:${HTSLIB_ROOT}/lib/pkgconfig:${HTSLIB_ROOT}/lib64/pkgconfig:$ENV{PKG_CONFIG_PATH}")
  pkg_check_modules(HTSLIB_PC QUIET IMPORTED_TARGET htslib)
endif()
set(HTSLIB_LINK_LIBRARIES "${HTSLIB_LIBRARY}" ZLIB::ZLIB m pthread)
if(TARGET PkgConfig::HTSLIB_PC)
  set(HTSLIB_LINK_LIBRARIES PkgConfig::HTSLIB_PC)
endif()

set(BWA_C_SOURCES ${BWA_API_DIR}/bwt.c ${BWA_API_DIR}/bntseq.c ${BWA_API_DIR}/utils.c)
set(FM_RANGE_SOURCES src/sa_range.cpp src/batmis_rank_backend.cpp src/bwa_fmd_index.cpp src/mismatch_search.cpp src/exact_search.cpp src/mismatch_pipeline.cpp)

# Default library remains the one used by tests.
add_library(bwa_bwt STATIC ${BWA_C_SOURCES})
target_include_directories(bwa_bwt PUBLIC ${BWA_API_DIR})
target_link_libraries(bwa_bwt PUBLIC ZLIB::ZLIB)
add_library(fm_range ${FM_RANGE_SOURCES})
target_include_directories(fm_range PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})
target_link_libraries(fm_range PUBLIC bwa_bwt ZLIB::ZLIB)

# Independent variants permit debug and optimized CLIs in the same build tree.
add_library(bwa_bwt_debug STATIC ${BWA_C_SOURCES})
target_include_directories(bwa_bwt_debug PUBLIC ${BWA_API_DIR})
target_link_libraries(bwa_bwt_debug PUBLIC ZLIB::ZLIB)
add_library(bwa_bwt_prod STATIC ${BWA_C_SOURCES})
target_include_directories(bwa_bwt_prod PUBLIC ${BWA_API_DIR})
target_link_libraries(bwa_bwt_prod PUBLIC ZLIB::ZLIB)
add_library(fm_range_debug STATIC ${FM_RANGE_SOURCES})
target_include_directories(fm_range_debug PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})
target_link_libraries(fm_range_debug PUBLIC bwa_bwt_debug ZLIB::ZLIB)
add_library(fm_range_prod STATIC ${FM_RANGE_SOURCES})
target_include_directories(fm_range_prod PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})
target_link_libraries(fm_range_prod PUBLIC bwa_bwt_prod ZLIB::ZLIB)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bwa_bwt_debug PRIVATE -O0 -g3 -DDEBUG)
  target_compile_options(fm_range_debug PRIVATE -O0 -g3 -DDEBUG)
  target_compile_options(bwa_bwt_prod PRIVATE -O3 -DNDEBUG)
  target_compile_options(fm_range_prod PRIVATE -O3 -DNDEBUG)
endif()
add_executable(fmsearch src/fmsearch.cpp src/sam_output.cpp)
target_include_directories(fmsearch PRIVATE ${HTSLIB_INCLUDE_DIR})
target_link_libraries(fmsearch PRIVATE fm_range_prod ${HTSLIB_LINK_LIBRARIES})
add_executable(fmsearch_debug src/fmsearch.cpp src/sam_output.cpp)
target_include_directories(fmsearch_debug PRIVATE ${HTSLIB_INCLUDE_DIR})
target_link_libraries(fmsearch_debug PRIVATE fm_range_debug ${HTSLIB_LINK_LIBRARIES})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fmsearch PRIVATE -O3 -DNDEBUG)
  target_compile_options(fmsearch_debug PRIVATE -O0 -g3 -DDEBUG)
endif()

# Existing test executables.
add_executable(test_sa_range tests/test_sa_range.cpp)
add_executable(test_bwa_extend_diagnostic tests/test_bwa_extend_diagnostic.cpp)
add_executable(test_exact_search tests/test_exact_search.cpp)

# This option is separate from BUILD_BENCHMARKS because it builds isolated
# exact-search benchmark variants without changing any regular target.
if(BUILD_SINGLETON_INLINE_BENCHMARKS)
  # Test-only build of the production run loop with per-step state capture.
  # Timed production fmsearch is compiled without trace code.
  add_library(fm_range_singleton_run_loop_trace STATIC ${FM_RANGE_SOURCES})
  target_include_directories(fm_range_singleton_run_loop_trace PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})

  target_link_libraries(fm_range_singleton_run_loop_trace PUBLIC bwa_bwt_prod ZLIB::ZLIB)
  target_compile_definitions(test_forward_only_search PRIVATE
    FM_BENCH_SINGLETON_RUN_LOOP_TRACE)
  target_link_libraries(test_forward_only_search PRIVATE fm_range_singleton_run_loop_trace)
  target_link_libraries(test_exact_search PRIVATE fm_range_prod)

  target_compile_definitions(fm_range_singleton_baseline PRIVATE FM_BENCH_SINGLETON_STEP_BASELINE)
  target_compile_options(fm_range_singleton_baseline PRIVATE -O3 -DNDEBUG)
  target_link_libraries(fm_range_singleton_baseline PUBLIC bwa_bwt_prod ZLIB::ZLIB)
  add_executable(fmsearch_singleton_baseline src/fmsearch.cpp src/sam_output.cpp)
  target_include_directories(fmsearch_singleton_baseline PRIVATE ${HTSLIB_INCLUDE_DIR})
  target_link_libraries(fmsearch_singleton_baseline PRIVATE fm_range_singleton_baseline ${HTSLIB_LINK_LIBRARIES})
  target_compile_options(fmsearch_singleton_baseline PRIVATE -O3 -DNDEBUG)

  add_executable(bench_singleton_inline_a benchmark/bench_singleton_inline.cpp)
  target_compile_definitions(bench_singleton_inline_a PRIVATE BENCH_VARIANT_NAME="A-baseline")
  target_link_libraries(bench_singleton_inline_a PRIVATE fm_range_singleton_baseline)

  add_library(fm_range_singleton_noinline STATIC ${FM_RANGE_SOURCES})
  target_include_directories(fm_range_singleton_noinline PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})
  target_compile_definitions(fm_range_singleton_noinline PRIVATE FM_BENCH_SINGLETON_NOINLINE)
  target_compile_options(fm_range_singleton_noinline PRIVATE -O3 -DNDEBUG)
  target_link_libraries(fm_range_singleton_noinline PUBLIC bwa_bwt_prod ZLIB::ZLIB)
  add_executable(bench_singleton_inline_b benchmark/bench_singleton_inline.cpp)
  target_compile_definitions(bench_singleton_inline_b PRIVATE FM_BENCH_SINGLETON_NOINLINE BENCH_VARIANT_NAME="B-noinline")
  target_link_libraries(bench_singleton_inline_b PRIVATE fm_range_singleton_noinline)

  add_library(fm_range_singleton_alwaysinline STATIC ${FM_RANGE_SOURCES})
  target_include_directories(fm_range_singleton_alwaysinline PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${BWA_API_DIR})
  target_compile_definitions(fm_range_singleton_alwaysinline PRIVATE FM_BENCH_SINGLETON_ALWAYS_INLINE)
  target_compile_options(fm_range_singleton_alwaysinline PRIVATE -O3 -DNDEBUG)
  target_link_libraries(fm_range_singleton_alwaysinline PUBLIC bwa_bwt_prod ZLIB::ZLIB)

  target_compile_definitions(bench_singleton_inline_c PRIVATE FM_BENCH_SINGLETON_ALWAYS_INLINE BENCH_VARIANT_NAME="C-always-inline")
  target_link_libraries(bench_singleton_inline_c PRIVATE fm_range_singleton_alwaysinline)

  add_executable(fmsearch_singleton_noinline src/fmsearch.cpp src/sam_output.cpp)
  target_include_directories(fmsearch_singleton_noinline PRIVATE ${HTSLIB_INCLUDE_DIR})
  target_link_libraries(fmsearch_singleton_noinline PRIVATE fm_range_singleton_noinline ${HTSLIB_LINK_LIBRARIES})
  target_compile_options(fmsearch_singleton_noinline PRIVATE -O3 -DNDEBUG)
  add_executable(fmsearch_singleton_alwaysinline src/fmsearch.cpp src/sam_output.cpp)
  target_include_directories(fmsearch_singleton_alwaysinline PRIVATE ${HTSLIB_INCLUDE_DIR})
  target_link_libraries(fmsearch_singleton_alwaysinline PRIVATE fm_range_singleton_alwaysinline ${HTSLIB_LINK_LIBRARIES})

enable_testing()
find_program(BWA_INDEX_EXECUTABLE NAMES bwa)
if(NOT BWA_INDEX_EXECUTABLE)
  message(FATAL_ERROR "CTest fixtures require the BWA executable (bwa index) on PATH")
endif()
set(TEST_FIXTURE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/test-fixtures")
set(EXACT_FIXTURE_DIR "${TEST_FIXTURE_ROOT}/exact")
set(LOCATE_FIXTURE_DIR "${TEST_FIXTURE_ROOT}/locate")
file(MAKE_DIRECTORY "${EXACT_FIXTURE_DIR}" "${LOCATE_FIXTURE_DIR}")
set(EXACT_FIXTURE_FASTA "${EXACT_FIXTURE_DIR}/reference.fa")
set(EXACT_FIXTURE_PREFIX "${EXACT_FIXTURE_DIR}/test")
file(WRITE "${EXACT_FIXTURE_FASTA}" ">test\nACGTTGCAACG\n")
execute_process(COMMAND "${BWA_INDEX_EXECUTABLE}" index -p "${EXACT_FIXTURE_PREFIX}" "${EXACT_FIXTURE_FASTA}" RESULT_VARIABLE EXACT_INDEX_RESULT OUTPUT_VARIABLE EXACT_INDEX_OUTPUT ERROR_VARIABLE EXACT_INDEX_ERROR)
if(NOT EXACT_INDEX_RESULT EQUAL 0 OR NOT EXISTS "${EXACT_FIXTURE_PREFIX}.bwt" OR NOT EXISTS "${EXACT_FIXTURE_PREFIX}.sa")
  message(FATAL_ERROR "Failed to generate exact-search BWA test index with ${BWA_INDEX_EXECUTABLE}:\n${EXACT_INDEX_OUTPUT}${EXACT_INDEX_ERROR}")
endif()
set(LOCATE_FIXTURE_FASTA "${LOCATE_FIXTURE_DIR}/reference.fa")
set(LOCATE_FIXTURE_PREFIX "${LOCATE_FIXTURE_DIR}/test")
file(WRITE "${LOCATE_FIXTURE_FASTA}" ">ref1\nACGTTGCAACG\n>ref2\nTTACCGGTAA\n")
execute_process(COMMAND "${BWA_INDEX_EXECUTABLE}" index -p "${LOCATE_FIXTURE_PREFIX}" "${LOCATE_FIXTURE_FASTA}" RESULT_VARIABLE LOCATE_INDEX_RESULT OUTPUT_VARIABLE LOCATE_INDEX_OUTPUT ERROR_VARIABLE LOCATE_INDEX_ERROR)
if(NOT LOCATE_INDEX_RESULT EQUAL 0 OR NOT EXISTS "${LOCATE_FIXTURE_PREFIX}.bwt" OR NOT EXISTS "${LOCATE_FIXTURE_PREFIX}.sa" OR NOT EXISTS "${LOCATE_FIXTURE_PREFIX}.ann" OR NOT EXISTS "${LOCATE_FIXTURE_PREFIX}.amb" OR NOT EXISTS "${LOCATE_FIXTURE_PREFIX}.pac")
  message(FATAL_ERROR "Failed to generate locate BWA test index with ${BWA_INDEX_EXECUTABLE}:\n${LOCATE_INDEX_OUTPUT}${LOCATE_INDEX_ERROR}")
endif()
add_test(NAME test_sa_range COMMAND test_sa_range)
add_test(NAME test_extend_diagnostics COMMAND test_bwa_extend_diagnostic "${EXACT_FIXTURE_PREFIX}.bwt")

add_test(NAME test_bwa_fmd_exhaustive COMMAND test_bwa_fmd_exhaustive "${EXACT_FIXTURE_PREFIX}" "${EXACT_FIXTURE_FASTA}")
